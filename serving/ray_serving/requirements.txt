# The versions a model is SERVED with. Every `==` pin here must equal the version in uv.lock,
# which is what the model was trained with — tests/unit/test_training_serving_versions_agree.py
# fails on any difference. Move a pin and the lock together, never one alone.
mlflow==3.16.0
ray[serve]==2.55.0
# <0.137: ray 2.55.0 @serve.ingress cloudpickles the FastAPI app; fastapi>=0.137 adds an
# unpicklable _thread.lock and breaks serve startup at import. 0.136.x is verified-compatible.
fastapi>=0.136.3,<0.137
# pandas and scikit-learn were floating (>=2.0, >=1.3): with mlflow 3.16 lifting its pandas<3
# cap, this image would have resolved pandas 3 while training locks 2.3.3 — a different major,
# with different copy and string-dtype semantics under the same feature transforms.
pandas==2.3.3
scikit-learn==1.9.1
xgboost==3.2.0
numpy
boto3>=1.43.88
# Open Inference Protocol v2 over gRPC (serving/oip_grpc, ADR 0126): its generated stubs need
# grpcio >= 1.80 and protobuf >= 6.31.1 at run time; httpx calls the REST implementation.
grpcio>=1.83.1
protobuf>=6.31.1,<7
httpx>=0.28.1
